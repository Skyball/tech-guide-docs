HTTP API 实战

实战：用 Go 写一个 HTTP server + JSON API（net/http + encoding/json）
实现 /alert 端点，接收 JSON 请求并返回 JSON 响应
90 分钟
package main

import (
    "fmt"
    "net/http"
    "encoding/json"
)

type AlertRequest struct {
    Message string `json:"message"`
    Level   string `json:"level"`
}

type AlertResponse struct {
    Status  string `json:"status"`
    AgentID string `json:"agent_id"`
}

func handleAlert(w http.ResponseWriter, r *http.Request) {
    var req AlertRequest
    json.NewDecoder(r.Body).Decode(&req)

    resp := AlertResponse{Status: "received", AgentID: "agent-001"}
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(resp)
}

func main() {
    http.HandleFunc("/alert", handleAlert)
    fmt.Println("Server running on :8080")
    http.ListenAndServe(":8080", nil)
}
# 测试
curl -X POST http://localhost:8080/alert \
  -H "Content-Type: application/json" \
  -d '{"message":"CPU 使用率 95%","level":"critical"}'
