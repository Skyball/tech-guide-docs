// proto/analyzer.proto
syntax = "proto3";
package analyzer;
option go_package = "./pb";

service Analyzer {
  rpc Analyze(AnalyzeRequest) returns (stream AnalyzeChunk);
}

message AnalyzeRequest { string alert_id = 1; string content = 2; }
message AnalyzeChunk   { string text = 1; bool done = 2; }

// 生成代码
// protoc --go_out=. --go-grpc_out=. proto/analyzer.proto

// server.go
func (s *Server) Analyze(req *pb.AnalyzeRequest, stream pb.Analyzer_AnalyzeServer) error {
    for _, chunk := range s.agent.Run(req.Content) {
        stream.Send(&pb.AnalyzeChunk{Text: chunk})
    }
    return stream.Send(&pb.AnalyzeChunk{Done: true})
}
