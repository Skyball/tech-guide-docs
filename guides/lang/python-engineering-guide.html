def format_alert(
    title: str,
    level: str = "warning",
    timestamp: str | None = None,
) -> str:

装饰器（面试必问）

import time
from functools import wraps

def timer(func):
    """测量函数执行时间的装饰器。"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.monotonic()
        result = func(*args, **kwargs)
        elapsed = time.monotonic() - start
        print(f"{func.__name__} took {elapsed:.3f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "done"

slow_function()  # "slow_function took 1.001s"
装饰器本质：接收一个函数，返回一个新函数。@timer 等价于 slow_function = timer(slow_function)。

题目 1：写一个装饰器 @retry(max_retries=3)，当被装饰函数抛异常时自动重试，超过次数则抛出原始异常。

参考答案 
from functools import wraps
import time

def retry(max_retries: int = 3, delay: float = 1.0):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception:
                    if attempt == max_retries - 1:
                        raise
                    time.sleep(delay * (2 ** attempt))
        return wrapper
    return decorator

@retry(max_retries=3, delay=0.5)
def flaky_api_call():
    import random
    if random.random() < 0.7:
        raise ConnectionError("timeout")
    return "success"

题目：设计一个 ToolRegistry 类，支持：(1) 用 @registry.register 装饰器注册工具函数；(2) 用 registry.get("tool_name") 获取工具；(3) 用 registry.list_tools() 列出所有已注册工具的名称和描述。

参考答案 
from dataclasses import dataclass, field
from typing import Callable

@dataclass
class ToolInfo:
    name: str
    description: str
    func: Callable

class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, ToolInfo] = {}

    def register(self, func: Callable) -> Callable:
        name = func.__name__
        desc = func.__doc__ or "No description"
        self._tools[name] = ToolInfo(name, desc.strip(), func)
        return func

    def get(self, name: str) -> Callable:
        if name not in self._tools:
            raise KeyError(f"Tool '{name}' not found")
        return self._tools[name].func

    def list_tools(self) -> list[dict]:
        return [{"name": t.name, "description": t.description}
                for t in self._tools.values()]

registry = ToolRegistry()

@registry.register
def search_orders(user_id: str):
    """根据用户 ID 查询订单列表。"""
    return [{"order_id": "123"}]

@registry.register
def create_refund(order_id: str, reason: str):
    """发起退款申请。"""
    return {"status": "submitted"}

class ToolError(Exception):
    def __init__(self, tool_name: str, message: str):
        self.tool_name = tool_name
        super().__init__(f"[{tool_name}] {message}")

class ToolNotFoundError(ToolError):
    pass

class ToolTimeoutError(ToolError):
    def __init__(self, tool_name: str, timeout_s: float):
        self.timeout_s = timeout_s
        super().__init__(tool_name, f"timed out after {timeout_s}s")

class ToolValidationError(ToolError):
    def __init__(self, tool_name: str, field: str, reason: str):
        self.field = field
        super().__init__(tool_name, f"invalid param '{field}': {reason}")

def execute_tool(name: str, params: dict) -> str:
    try:
        tool = registry.get(name)
        return tool(**params)
    except ToolNotFoundError as e:
        return f"抱歉，功能 {e.tool_name} 不可用"
    except ToolTimeoutError as e:
        return f"操作超时（{e.timeout_s}s），请稍后重试"
    except ToolValidationError as e:
        return f"参数错误：{e.field} - {e}"
    except ToolError:
        return "工具执行出错，请稍后重试"

题目：写一个异步函数 batch_call_model，接收一组 prompt，用 asyncio.Semaphore 限制最多 3 个并发，用 asyncio.as_completed 实现"谁先完成先处理谁"。模拟每次调用耗时 0.5-2 秒。

import asyncio
import random

sem = asyncio.Semaphore(3)

async def call_model(prompt: str) -> str:
    async with sem:
        delay = random.uniform(0.5, 2.0)
        await asyncio.sleep(delay)
        return f"[{delay:.1f}s] Response to: {prompt[:20]}"

async def batch_call_model(prompts: list[str]) -> list[str]:
    tasks = [asyncio.create_task(call_model(p)) for p in prompts]
    results = []
    for coro in asyncio.as_completed(tasks):
        result = await coro
        print(f"Done: {result}")
        results.append(result)
    return results

# 测试

    prompts = [f"Question {i}" for i in range(8)]
    results = await batch_call_model(prompts)
    print(f"Total: {len(results)} results")

    func: Callable[..., T],

) -> T:

import asyncio

async def fetch(url: str) -> str:
    await asyncio.sleep(0.5)
    if "bad" in url:
        raise ValueError(f"Bad URL: {url}")
    return f"OK: {url}"

async def worker(name: str, queue: asyncio.Queue, results: list):
    while True:
        url = await queue.get()
        try:
            result = await fetch(url)
            results.append(result)
            print(f"{name}: {result}")
        except Exception as e:
            print(f"{name} error: {e}")
        finally:
            queue.task_done()

async def main():
    queue = asyncio.Queue()
    results = []
    urls = [f"https://api.com/{i}" for i in range(8)]
    urls += ["https://bad.com/1", "https://bad.com/2"]

    for url in urls:
        queue.put_nowait(url)

    workers = [
        asyncio.create_task(worker(f"W{i}", queue, results))
        for i in range(3)
    ]

    await queue.join()  # 等所有任务 task_done
    for w in workers:
        w.cancel()       # 优雅关闭 worker
    print(f"Done: {len(results)} successes")

asyncio.run(main())

print(sys.getsizeof([]))        # 56 bytes（空 list）

print(sys.getsizeof("hello"))   # 54 bytes
print(sys.getsizeof(42))        # 28 bytes（int）

# 递归计算对象总大小（含引用的子对象）
def deep_sizeof(obj, seen=None):
    if seen is None:
        seen = set()
    obj_id = id(obj)
    if obj_id in seen:
        return 0
    seen.add(obj_id)
    size = sys.getsizeof(obj)
    if isinstance(obj, dict):
        size += sum(deep_sizeof(k, seen) + deep_sizeof(v, seen)
                    for k, v in obj.items())
    elif isinstance(obj, (list, tuple, set, frozenset)):
        size += sum(deep_sizeof(i, seen) for i in obj)
    return size

import timeit
import random
import string

def random_query():
    return "".join(random.choices(string.ascii_lowercase, k=8))

def dedup_list(items):
    seen = []
    result = []
    for item in items:
        if item not in seen:
            seen.append(item)
            result.append(item)
    return result

# 快：用 set 做 in 检查 → O(n)
def dedup_set(items):
    seen = set()
    result = []
    for item in items:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

    items.append(item)
    return items
GIL（全局解释器锁）

CPython 的 GIL 让同一时刻只有一个线程执行 Python 字节码。

影响：CPU 密集型任务多线程无法加速。IO 密集型（网络请求）不受影响——线程在等 IO 时释放 GIL。

AI Agent 的做法：用 asyncio（协程）处理 IO 密集型任务（模型 API 调用），用 multiprocessing 处理 CPU 密集型任务（embedding 计算）。

c = copy.deepcopy(a)  # 深拷贝：完全独立

print(b[0])  # [1, 2, 99] — 浅拷贝受影响
print(c[0])  # [1, 2]    — 深拷贝不受影响
列表推导 vs 生成器表达式 
# 列表推导：一次性生成所有结果（占内存）
squares = [x**2 for x in range(1000000)]

# 生成器表达式：惰性求值（省内存）

数据量小用列表推导（简洁），数据量大用生成器（省内存）。

__init__ vs __new__ vs __call__
__new__创建实例（分配内存）极少重写
__init__初始化实例（设置属性）最常用
__call__让实例可以像函数一样调用实现中间件、装饰器
asyncio 事件循环

事件循环是异步编程的核心——一个 while 循环不断检查有没有任务可以执行。

await 的意思是"这里要等 IO，先去执行别的协程"。事件循环在 IO 完成后把控制权交回来。

关键：asyncio.run() 创建事件循环并运行。FastAPI 自动管理事件循环，你只管 async def + await。

from functools import wraps
from typing import Callable, Any

class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, Callable] = {}

    def tool(self, description: str):
        def decorator(func: Callable) -> Callable:
            @wraps(func)
            async def wrapper(*args, **kwargs) -> ToolResult:
                try:
                    result = await func(*args, **kwargs)
                    return ToolResult(func.__name__, True, str(result))
                except Exception as e:
                    return ToolResult(func.__name__, False, f"Error: {e}")
            wrapper._description = description
            self._tools[func.__name__] = wrapper
            return wrapper
        return decorator

    async def call(self, name: str, **kwargs) -> ToolResult:
        if name not in self._tools:
            return ToolResult(name, False, f"Tool '{name}' not found")
        return await self._tools[name](**kwargs)

    def list_tools(self) -> list[dict]:
        return [
            {"name": n, "description": f._description}
            for n, f in self._tools.items()
        ]

Step 3：实现 Agent 核心循环

Agent 是一个 while 循环：Think → Act → Respond。用异步生成器实现流式输出。

import asyncio
import re

class MiniAgent:
    def __init__(self, tools: ToolRegistry):
        self.tools = tools
        self.max_steps = 5

    async def _think(self, state: AgentState) -> tuple[str, dict | None]:
        """模拟 LLM 决策：是否调用工具。"""
        last_msg = state["messages"][-1]["content"].lower()

        if "订单" in last_msg or "order" in last_msg:
            match = re.search(r"(ORD\d+)", last_msg, re.IGNORECASE)
            order_id = match.group(1) if match else "ORD001"
            return "query_order", {"order_id": order_id}

        if "退款" in last_msg or "refund" in last_msg:
            return "create_refund", {"order_id": "ORD001", "reason": "用户申请"}

        return "respond", None

    async def _respond(self, state: AgentState) -> str:
        """模拟 LLM 生成回复。"""
        tool_results = [
            m["content"] for m in state["messages"]
            if m["role"] == "tool"
        ]
        user_msg = state["messages"][0]["content"]

        if tool_results:
            return f"根据查询结果，{tool_results[-1]}。请问还需要其他帮助吗？"
        return f"您好！关于「{user_msg}」，我是智能客服，请问有什么可以帮您？"

    async def run(self, user_message: str):
        """Agent 主循环，异步生成器逐步返回过程。"""
        state: AgentState = {
            "messages": [{"role": "user", "content": user_message}],
            "tool_calls": [],
            "current_step": "think",
        }

        for step in range(self.max_steps):
            if state["current_step"] == "think":
                tool_name, params = await self._think(state)
                if tool_name == "respond":
                    state["current_step"] = "respond"
                else:
                    yield f"🔧 调用工具: {tool_name}({params})\n"
                    result = await self.tools.call(tool_name, **params)
                    state["tool_calls"].append(tool_name)
                    state["messages"].append({
                        "role": "tool",
                        "content": result.output,
                    })
                    status = "✅" if result.success else "❌"
                    yield f"{status} 工具结果: {result.output}\n"
                    state["current_step"] = "respond"

            if state["current_step"] == "respond":
                answer = await self._respond(state)
                for char in answer:
                    yield char
                    await asyncio.sleep(0.02)  # 模拟逐字输出
                state["current_step"] = "done"
                break

from functools import lru_cache
from collections import OrderedDict
import time

class TTLCache:
    """带 TTL 和容量上限的缓存。"""
    def __init__(self, maxsize: int = 200, ttl: int = 3600):
        self._cache: OrderedDict = OrderedDict()
        self._maxsize = maxsize
        self._ttl = ttl

    def get(self, key: str) -> list | None:
        if key in self._cache:
            val, ts = self._cache[key]
            if time.time() - ts < self._ttl:
                self._cache.move_to_end(key)
                return val
            del self._cache[key]
        return None

    def set(self, key: str, value: list):
        self._cache[key] = (value, time.time())
        if len(self._cache) > self._maxsize:
            self._cache.popitem(last=False)  # 淘汰最老

session_cache = TTLCache(maxsize=200, ttl=3600)

import asyncio

class RateLimiter:
    """令牌桶限流器。"""
    def __init__(self, rpm: int = 50):
        self._sem = asyncio.Semaphore(rpm)
        self._rpm = rpm

    async def acquire(self):
        await self._sem.acquire()
        asyncio.get_running_loop().call_later(
            60.0 / self._rpm, self._sem.release
        )

limiter = RateLimiter(rpm=50)

async def call_model_safe(prompt: str, max_retries=3):
    for attempt in range(max_retries):
        await limiter.acquire()
        try:
            return await client.post(url, json={"prompt": prompt}, timeout=30)
        except httpx.HTTPStatusError as e:
            if e.response.status_code == 429:
                wait = float(e.response.headers.get("Retry-After", 2 ** attempt))
                await asyncio.sleep(wait)
            else:
                raise
    raise MaxRetriesError(f"Failed after {max_retries} retries")

        *[call_model_safe(a) for a in alerts]

async def process_refund_v1(user_id, amount):

async def process_refund_v2(user_id, amount, max_retries=3):

            "SELECT balance, version FROM accounts WHERE user_id = $1",
            user_id

P0 事故 10：消息队列消费积压导致 Agent 分析延迟 2 小时

现象
  • 告警：消息队列 Lag 从 0 增长到 50000+，用户提交的告警分析请求 2 小时没有响应
  • 监控：Consumer 消费速度从 100 msg/s 下降到 2 msg/s
  • 日志：大量 TimeoutError: Model API timeout after 60s
  • 触发条件：模型供应商侧故障，API 延迟从 2s 增加到 60s（超时）
根因分析

Consumer 同步处理每条消息：接收消息 → 调模型 API → 等结果 → ack。当模型 API 变慢（2s → 60s），消费速度线性下降 30 倍。加上失败重试，实际消费速度降到 2 msg/s，远低于生产速度。积压越来越多。
